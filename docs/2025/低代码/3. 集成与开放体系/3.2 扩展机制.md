# 3.2 扩展机制

## 3.2.1 组件开发规范（Props定义、生命周期）

### 业务逻辑事件扩展
- **事件定义**：
  - 定义标准化的业务事件接口
  - 支持事件的发布和订阅
  - 提供事件参数的类型定义

- **事件扩展机制**：
  - 使用自定义Hooks封装事件逻辑
  - 支持事件的注册和注销
  - 提供事件触发和处理的机制

- **示例**：
```typescript
import React, { createContext, useContext, useRef, useEffect, useState } from 'react';

// 事件系统核心
interface Event {
  name: string;
  payload: any;
}

interface EventSystemContextType {
  on: (eventName: string, handler: (payload: any) => void) => void;
  off: (eventName: string, handler: (payload: any) => void) => void;
  emit: (eventName: string, payload: any) => void;
}

const EventSystemContext = createContext<EventSystemContextType | undefined>(undefined);

// 事件系统提供者
export const EventSystemProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const eventHandlersRef = useRef<Record<string, Array<(payload: any) => void>>({});

  // 注册事件处理器
  const on = (eventName: string, handler: (payload: any) => void) => {
    if (!eventHandlersRef.current[eventName]) {
      eventHandlersRef.current[eventName] = [];
    }
    eventHandlersRef.current[eventName].push(handler);
  };

  // 注销事件处理器
  const off = (eventName: string, handler: (payload: any) => void) => {
    if (eventHandlersRef.current[eventName]) {
      eventHandlersRef.current[eventName] = eventHandlersRef.current[eventName].filter(
        h => h !== handler
      );
    }
  };

  // 触发事件
  const emit = (eventName: string, payload: any) => {
    if (eventHandlersRef.current[eventName]) {
      eventHandlersRef.current[eventName].forEach(handler => {
        try {
          handler(payload);
        } catch (error) {
          console.error(`Error in event handler for ${eventName}:`, error);
        }
      });
    }
  };

  return (
    <EventSystemContext.Provider value={{ on, off, emit }}>
      {children}
    </EventSystemContext.Provider>
  );
};

// 使用事件系统的Hook
export const useEventSystem = () => {
  const context = useContext(EventSystemContext);
  if (!context) {
    throw new Error('useEventSystem must be used within an EventSystemProvider');
  }
  return context;
};

// 字段级生命周期事件示例
const FieldService = () => {
  const { 
    emitFieldValueChanged, 
    emitFieldVisibility, 
    emitFieldEnabled, 
    emitFieldRequired 
  } = useBusinessEvents();

  // 处理字段值变化
  const handleFieldChange = (field: string, value: any, oldValue: any) => {
    console.log(`Field ${field} changed from ${oldValue} to ${value}`);
    
    // 触发字段值变化事件
    emitFieldValueChanged(field, value, oldValue);
    
    // 根据字段值变化，触发其他字段的状态变化
    if (field === 'role') {
      // 当角色变化时，控制部门和薪资字段的可见性和必填性
      emitFieldVisibility('department', { role: value });
      emitFieldVisibility('salary', { role: value });
      emitFieldRequired('department', { role: value });
      emitFieldRequired('salary', { role: value });
    }
  };

  // 检查字段可见性
  const checkFieldVisibility = (field: string, context: any) => {
    // 根据上下文判断字段是否可见
    if (field === 'department') {
      return context.role === 'admin' || context.role === 'user';
    } else if (field === 'salary') {
      return context.role === 'admin';
    }
    return true;
  };

  // 检查字段是否启用
  const checkFieldEnabled = (field: string, context: any) => {
    // 根据上下文判断字段是否启用
    if (field === 'salary' && context.role !== 'admin') {
      return false;
    }
    return true;
  };

  // 检查字段是否必填
  const checkFieldRequired = (field: string, context: any) => {
    // 根据上下文判断字段是否必填
    if (field === 'department' && (context.role === 'admin' || context.role === 'user')) {
      return true;
    } else if (field === 'salary' && context.role === 'admin') {
      return true;
    }
    return false;
  };

  return {
    handleFieldChange,
    checkFieldVisibility,
    checkFieldEnabled,
    checkFieldRequired
  };
};
```

### 钩子机制
- **基于React Hooks的插件系统**：
  - 使用自定义Hooks实现插件的挂载和生命周期管理
  - 支持插件间的通信和数据共享

- **示例**：
```typescript
// 插件系统核心
import React, { createContext, useContext, useRef, useEffect, useState } from 'react';

// 插件上下文
interface Plugin {
  id: string;
  name: string;
  version: string;
  activate: () => void;
  deactivate: () => void;
  hooks?: Record<string, Function>;
}

interface PluginContextType {
  plugins: Plugin[];
  registerPlugin: (plugin: Plugin) => void;
  unregisterPlugin: (pluginId: string) => void;
  callHook: (hookName: string, ...args: any[]) => any[];
}

const PluginContext = createContext<PluginContextType | undefined>(undefined);

// 插件系统提供者
export const PluginProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [plugins, setPlugins] = useState<Plugin[]>([]);
  const pluginsRef = useRef<Plugin[]>([]);

  // 注册插件
  const registerPlugin = (plugin: Plugin) => {
    setPlugins(prev => {
      const updated = [...prev, plugin];
      pluginsRef.current = updated;
      plugin.activate();
      return updated;
    });
  };

  // 卸载插件
  const unregisterPlugin = (pluginId: string) => {
    setPlugins(prev => {
      const plugin = prev.find(p => p.id === pluginId);
      if (plugin) {
        plugin.deactivate();
      }
      const updated = prev.filter(p => p.id !== pluginId);
      pluginsRef.current = updated;
      return updated;
    });
  };

  // 调用钩子
  const callHook = (hookName: string, ...args: any[]) => {
    const results: any[] = [];
    pluginsRef.current.forEach(plugin => {
      if (plugin.hooks && plugin.hooks[hookName]) {
        try {
          const result = plugin.hooks[hookName](...args);
          results.push(result);
        } catch (error) {
          console.error(`Error in plugin ${plugin.id} hook ${hookName}:`, error);
        }
      }
    });
    return results;
  };

  return (
    <PluginContext.Provider value={{ plugins, registerPlugin, unregisterPlugin, callHook }}>
      {children}
    </PluginContext.Provider>
  );
};

// 使用插件系统的Hook
export const usePlugins = () => {
  const context = useContext(PluginContext);
  if (!context) {
    throw new Error('usePlugins must be used within a PluginProvider');
  }
  return context;
};

// 插件示例
const ExamplePlugin: Plugin = {
  id: 'example-plugin',
  name: 'Example Plugin',
  version: '1.0.0',
  activate: () => {
    console.log('Example Plugin activated');
  },
  deactivate: () => {
    console.log('Example Plugin deactivated');
  },
  hooks: {
    'beforeRender': (component: string) => {
      console.log(`Before rendering ${component}`);
      return `Processed by Example Plugin: ${component}`;
    },
    'afterRender': (component: string, result: any) => {
      console.log(`After rendering ${component}:`, result);
    }
  }
};

// 使用插件系统
const App: React.FC = () => {
  const { registerPlugin, callHook } = usePlugins();

  useEffect(() => {
    // 注册插件
    registerPlugin(ExamplePlugin);

    // 清理
    return () => {
      // 这里可以卸载插件
    };
  }, [registerPlugin]);

  const handleRender = () => {
    // 调用beforeRender钩子
    const beforeResults = callHook('beforeRender', 'MyComponent');
    console.log('Before render results:', beforeResults);

    // 模拟渲染
    const renderResult = 'Rendered content';

    // 调用afterRender钩子
    callHook('afterRender', 'MyComponent', renderResult);
  };

  return (
    <div>
      <h1>Plugin System Example</h1>
      <button onClick={handleRender}>Render Component</button>

  );
};

export default App;
```

### 依赖管理
- **插件依赖处理**：
  - 支持插件间的依赖声明和解析
  - 自动处理依赖的加载顺序
  - 检测和处理依赖冲突

- **示例**：
```typescript
// 插件依赖管理
import React, { useState, useEffect } from 'react';

interface PluginWithDependencies extends Plugin {
  dependencies?: string[];
}

const DependencyManager: React.FC = () => {
  const [plugins, setPlugins] = useState<PluginWithDependencies[]>([]);
  const [loadedPlugins, setLoadedPlugins] = useState<string[]>([]);

  // 解析依赖并加载插件
  const loadPluginsWithDependencies = async (pluginsToLoad: PluginWithDependencies[]) => {
    const loadOrder: string[] = [];
    const visited = new Set<string>();
    const temp = new Set<string>();

    // 拓扑排序确定加载顺序
    const topologicalSort = (pluginId: string) => {
      if (temp.has(pluginId)) {
        throw new Error(`Circular dependency detected for plugin ${pluginId}`);
      }
      if (visited.has(pluginId)) {
        return;
      }

      temp.add(pluginId);
      const plugin = pluginsToLoad.find(p => p.id === pluginId);
      if (plugin?.dependencies) {
        for (const depId of plugin.dependencies) {
          topologicalSort(depId);
        }
      }
      temp.delete(pluginId);
      visited.add(pluginId);
      loadOrder.push(pluginId);
    };

    // 对所有插件进行拓扑排序
    for (const plugin of pluginsToLoad) {
      if (!visited.has(plugin.id)) {
        try {
          topologicalSort(plugin.id);
        } catch (error) {
          console.error('Dependency resolution error:', error);
          return;
        }
      }
    }

    // 按顺序加载插件
    for (const pluginId of loadOrder) {
      const plugin = pluginsToLoad.find(p => p.id === pluginId);
      if (plugin) {
        try {
          // 这里可以添加实际的插件加载逻辑
          console.log(`Loading plugin ${plugin.name} (${plugin.id})`);
          plugin.activate();
          setLoadedPlugins(prev => [...prev, plugin.id]);
        } catch (error) {
          console.error(`Error loading plugin ${plugin.id}:`, error);
        }
      }
    }
  };

  useEffect(() => {
    // 示例插件
    const examplePlugins: PluginWithDependencies[] = [
      {
        id: 'base-plugin',
        name: 'Base Plugin',
        version: '1.0.0',
        activate: () => console.log('Base Plugin activated'),
        deactivate: () => console.log('Base Plugin deactivated')
      },
      {
        id: 'feature-plugin',
        name: 'Feature Plugin',
        version: '1.0.0',
        dependencies: ['base-plugin'],
        activate: () => console.log('Feature Plugin activated'),
        deactivate: () => console.log('Feature Plugin deactivated')
      },
      {
        id: 'advanced-plugin',
        name: 'Advanced Plugin',
        version: '1.0.0',
        dependencies: ['feature-plugin'],
        activate: () => console.log('Advanced Plugin activated'),
        deactivate: () => console.log('Advanced Plugin deactivated')
      }
    ];

    setPlugins(examplePlugins);
    loadPluginsWithDependencies(examplePlugins);
  }, []);

  return (
    <div>
      <h2>Dependency Manager</h2>
      <h3>Loaded Plugins:</h3>
      <ul>
        {loadedPlugins.map(pluginId => {
          const plugin = plugins.find(p => p.id === pluginId);
          return <li key={pluginId}>{plugin?.name} (v{plugin?.version})</li>;
        })}
      </ul>
    </div>
  );
};

export default DependencyManager;
```
